62/// Per-collector timing for a single stat collection run.
63///
64/// `start_offset_ms` is the wall time between the top of the collection cycle and
65/// the moment this collector's future was first polled. A non-trivial offset
66/// indicates the spawn/scheduling loop or the runtime is delaying first poll,
67/// which means collectors are not starting in parallel as intended.
68///
69/// `elapsed_ms` is the wall time between first poll and completion.
70#[derive(serde::Serialize, serde::Deserialize, Clone, Debug, Default, PartialEq)]
71pub struct CollectorTiming {
72    /// Name of the collector (e.g. "units", "pid1", "dbus_stats")
73    pub name: String,
74    /// Milliseconds from top of the run until the spawned future's first poll.
75    /// Should be small (< a few ms) when collectors are truly running in parallel.
76    pub start_offset_ms: f64,
77    /// Milliseconds from first poll to future completion.
78    pub elapsed_ms: f64,
79    /// Whether the collector returned Ok.
80    pub success: bool,
81}

83/// Which API transport served a collector on the last run.
84///
85/// Recorded per enabled collector so operators can watch varlink adoption
86/// climb as the fleet's systemd upgrades past each endpoint's minimum
87/// version: a collector reports `Varlink` when its varlink attempt succeeded
88/// and `Dbus` when it collected the legacy way — either because varlink is
89/// disabled in config or because the varlink attempt failed and fell back.
90/// `Dbus` also covers the file-based networkd fallback: the value answers
91/// "did this collector use varlink", not "which fallback served it".
92/// Collectors with no varlink path (`pid1`, `dbus_stats`) and disabled
93/// collectors stay `None` and emit no gauge, so the present gauges are
94/// exactly the enabled set — no separate enabled-collectors counter needed.
95#[derive(
96    serde_repr::Serialize_repr,
97    serde_repr::Deserialize_repr,
98    Clone,
99    Copy,
100    Debug,
101    Default,
102    PartialEq,
103    Eq,
104)]
105#[repr(u8)]
106pub enum CollectorTransport {
107    #[default]
108    Dbus = 0,
109    Varlink = 1,
110}
111
112impl CollectorTransport {
113    /// Gauge value for flat JSON and metric exporters: 1 when varlink served
114    /// the collector, 0 for the D-Bus/file fallback.
115    ///
116    /// The `repr(u8)` discriminants are the gauge values, so every output
117    /// format (json, json-pretty, json-flat) reports the same integer —
118    /// matching the other stat enums (`SystemdSystemState`, the unit state
119    /// enums) rather than serializing as a string in some formats.
120    pub fn as_u64(self) -> u64 {
121        self as u64
122    }
123}

125/// Per-collector transport record for the last collection run.
126///
127/// Every field is `Some` when its collector ran and `None` when it did not
128/// (disabled in config). Absent-when-disabled is what makes the adoption
129/// ratio work: `count()` over the gauges is the enabled set.
130///
131/// NOTE for future collectors: the host `MachineStats` is constructed once
132/// outside the daemon loop, not fresh each iteration — so a collector with
133/// an early-return path that skips its gauge assignment would silently
134/// inherit the previous run's value instead of dropping the gauge. Always
135/// assign on every path, including cache hits and config-disabled fallbacks.
136#[derive(serde::Serialize, serde::Deserialize, Clone, Debug, Default, PartialEq)]
137pub struct VarlinkUsage {
138    /// Always collected; follows `[system-state] varlink` via the shared
139    /// `Manager.Describe` call. Normally agrees with `system_state` (one
140    /// call serves both), but persists when `[system-state]` is disabled —
141    /// making it the cheapest varlink canary on such hosts.
142    #[serde(skip_serializing_if = "Option::is_none")]
143    pub version: Option<CollectorTransport>,
144    #[serde(skip_serializing_if = "Option::is_none")]
145    pub system_state: Option<CollectorTransport>,
146    /// `Varlink` means the bulk enumeration came from the metrics stream.
147    /// Inside containers this still includes D-Bus calls underneath: the
148    /// timer backfill (`collect_all_timers_dbus`) and the oneshot type
149    /// override, which have no varlink equivalent there (see #211). The host
150    /// varlink path needs neither, so a container `1` involves strictly more
151    /// D-Bus traffic than a host `1` — compare host and container gauges
152    /// separately rather than aggregating them into one adoption ratio.
153    #[serde(skip_serializing_if = "Option::is_none")]
154    pub units: Option<CollectorTransport>,
155    #[serde(skip_serializing_if = "Option::is_none")]
156    pub networkd: Option<CollectorTransport>,
157    /// Host-side machine enumeration, still D-Bus-only until machined grows
158    /// a varlink List API (see #37). Per-container transports land on each
159    /// machine's own `MachineStats` instead.
160    #[serde(skip_serializing_if = "Option::is_none")]
161    pub machines: Option<CollectorTransport>,
162    #[serde(skip_serializing_if = "Option::is_none")]
163    pub boot_blame: Option<CollectorTransport>,
164    #[serde(skip_serializing_if = "Option::is_none")]
165    pub verify: Option<CollectorTransport>,
166}

251/// Output produced by every spawned collector future after wrapping with timing.
252type TimedCollectorOutput = (String, anyhow::Result<()>, Duration, Duration);
253
254/// Spawn a collector future onto the join set with timing instrumentation.
255///
256/// The wrapping closure records the moment the future is first polled (relative
257/// to `collect_start`) and the elapsed wall time until it completes. Both
258/// durations and the collector name are returned alongside the original result.
259///
260/// `tokio::task::JoinSet::spawn` runs the future on a new task, and tracing
261/// spans do not cross task boundaries automatically — without explicitly
262/// capturing and re-attaching the caller's span here, every collector (and
263/// anything it spawns in turn, e.g. units.rs's per-unit tasks) would show up
264/// as an unrelated root trace instead of a child of the current collection run.
265fn spawn_timed<F>(
266    join_set: &mut tokio::task::JoinSet<TimedCollectorOutput>,
267    name: &'static str,
268    collect_start: Instant,
269    fut: F,
270) where
271    F: std::future::Future<Output = anyhow::Result<()>> + Send + 'static,
272{
273    let parent_span = tracing::Span::current();
274    let span = tracing::debug_span!(
275        parent: &parent_span,
276        "collector",
277        name,
278        elapsed_ms = tracing::field::Empty,
279        success = tracing::field::Empty,
280    );
281    let recording_span = span.clone();
282    join_set.spawn(
283        async move {
284            let task_first_poll = Instant::now();
285            let start_offset = task_first_poll.duration_since(collect_start);
286            let result = fut.await;
287            let elapsed = task_first_poll.elapsed();
288            recording_span.record("elapsed_ms", elapsed.as_secs_f64() * 1000.0);
289            recording_span.record("success", result.is_ok());
290            (name.to_string(), result, start_offset, elapsed)
291        }
292        .instrument(span),
293    );
294}

296/// Lazily-created system bus connection shared by every collector that
297/// needs D-Bus — also passed into collectors that decide their transport
298/// internally (`verify`, `boot_blame`), so those resolve it only on their
299/// D-Bus code paths rather than up front (see finding 1 on #224).
300///
301/// The cell starts empty (or seeded from `maybe_connection` for library
302/// callers) and is only connected on first use, so a run whose enabled
303/// collectors all succeed over varlink/fs/procfs never touches the bus at
304/// all — the prerequisite for the zero-D-Bus claim in #37.
305///
306/// `tokio::sync::OnceCell` serializes initializers behind a semaphore:
307/// concurrent first-users wait for the in-flight attempt and share its
308/// result — there is no race and no spare connection. A failed attempt is
309/// *not* cached (a waiter starts a fresh attempt instead), so a dead bus
310/// costs one connect attempt per D-Bus collector, sequentially. `ENOENT`
311/// fails fast and the run finishes in milliseconds, but `method_timeout`
312/// only bounds method calls, not `Builder::build()` — a socket that
313/// exists and hangs (wedged broker) would serialize one blocking connect
314/// per D-Bus collector per cycle, where the old code blocked exactly once
315/// at startup. Worth knowing when pointing at an unfamiliar bus address.
316pub(crate) type DbusCell = Arc<tokio::sync::OnceCell<zbus::Connection>>;
317
318/// Resolve the shared D-Bus connection, connecting on first use.
319///
320/// Only called on D-Bus code paths (varlink/fs-first collectors reach here
321/// solely through their fallbacks), so a missing bus surfaces as that
322/// collector's ordinary error — logged per-collector, never fatal to the
323/// run. Daemon-mode reuse is preserved: the same connection serves every
324/// cycle until a failed cycle drops it (see below).
325pub(crate) async fn dbus_connection(
326    cell: &DbusCell,
327    dbus_timeout: u64,
328) -> Result<zbus::Connection, MonitordError> {
329    cell.get_or_try_init(|| async {
330        zbus::connection::Builder::system()?
331            .method_timeout(std::time::Duration::from_secs(dbus_timeout))
332            .build()
333            .await
334    })
335    .await
336    .cloned()
337    .map_err(MonitordError::ZbusError)
338}

340/// Main statistic collection function running what's required by configuration in parallel
341/// Takes an optional locked stats struct to update and to output stats to STDOUT or not.
342/// Takes an optional D-Bus connection. Returns `Some(connection)` if the
343/// collection cycle completed without errors (meaning the connection is reusable),
344/// `None` if errors occurred.
345pub async fn stat_collector(
346    config: config::Config,
347    maybe_locked_stats: Option<Arc<RwLock<MonitordStats>>>,
348    output_stats: bool,
349    maybe_connection: Option<zbus::Connection>,
350) -> Result<Option<zbus::Connection>, MonitordError> {
351    let mut collect_interval_ms: u128 = 0;
352    if config.monitord.daemon {
353        collect_interval_ms = (config.monitord.daemon_stats_refresh_secs * 1000).into();
354    }
355
356    let config = Arc::new(config);
357    let locked_monitord_stats: Arc<RwLock<MonitordStats>> =
358        maybe_locked_stats.unwrap_or(Arc::new(RwLock::new(MonitordStats::default())));
359    let locked_machine_stats: Arc<RwLock<MachineStats>> =
360        Arc::new(RwLock::new(MachineStats::default()));
361    let cached_machine_connections: Arc<tokio::sync::Mutex<machines::MachineConnections>> =
362        Arc::new(tokio::sync::Mutex::new(HashMap::new()));
363
364    std::env::set_var("DBUS_SYSTEM_BUS_ADDRESS", &config.monitord.dbus_address);
365    // Seeded when the library caller passes a connection, otherwise
366    // connected lazily by the first collector that needs the bus. Mutable
367    // so a failed daemon cycle can swap in a fresh cell (see below) rather
368    // than letting the next cycle reuse a bus that may have gone away.
369    let mut dbus_cell: DbusCell = Arc::new(tokio::sync::OnceCell::new());
370    if let Some(conn) = maybe_connection {
371        // Seeding cannot fail: a fresh cell is always empty.
372        let _ = dbus_cell.set(conn);
373    }
374    let dbus_timeout = config.monitord.dbus_timeout;
375    let mut join_set: tokio::task::JoinSet<TimedCollectorOutput> = tokio::task::JoinSet::new();
376    let mut had_error;
377
378    loop {
379        let collect_start_time = Instant::now();
380        // Kept alive for the whole iteration so its reported duration covers
381        // the full run (spawn + drain), not just the synchronous spawn phase
382        // below where `run_guard` is held.
383        let run_span = tracing::info_span!("stat_collector_run");
384        let run_guard = run_span.enter();
385        info!("Starting stat collection run");
386
387        // One Manager.Describe serves both the version and system state
388        // collectors: PID 1 handles varlink requests one at a time, so a second
389        // call would sit behind the units collector's whole metrics stream.
390        //
391        // Which request reaches PID 1 first is still up to the scheduler, so
392        // under CPU pressure this call can land behind that stream anyway. That
393        // is the old cost for one call rather than two, not a new failure mode.
394        // Making it deterministic would mean holding the units collector until
395        // this resolves, which would put the collector that gates the cycle
396        // behind an unrelated socket.
397        let manager_describe = config.use_varlink(&[config.system_state.varlink]).then(|| {
398            crate::varlink_system::shared_describe(
399                crate::varlink_system::MANAGER_SOCKET_PATH.to_string(),
400            )
401        });
402
403        // Always collect systemd version
404        {
405            let dbus_cell = Arc::clone(&dbus_cell);
406            let stats_clone = locked_machine_stats.clone();
407            let describe = manager_describe.clone();
408            let no_fallback = config.varlink.no_fallback;
409            spawn_timed(&mut join_set, "version", collect_start_time, async move {
410                if let Some(describe) = describe {
411                    match crate::varlink_system::update_version(describe, stats_clone.clone()).await
412                    {
413                        Ok(()) => {
414                            stats_clone.write().await.varlink_usage.version =
415                                Some(CollectorTransport::Varlink);
416                            return Ok(());
417                        }
418                        Err(err) => {
419                            crate::varlink_fallback::report_varlink_failure(
420                                no_fallback,
421                                "version",
422                                "D-Bus",
423                                err,
424                            )?;
425                        }
426                    }
427                }
428                stats_clone.write().await.varlink_usage.version = Some(CollectorTransport::Dbus);
429                let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
430                crate::system::update_version(conn, stats_clone.clone()).await
431            });
432        }
433
434        // Collect pid1 procfs stats
435        if config.pid1.enabled {
436            spawn_timed(
437                &mut join_set,
438                "pid1",
439                collect_start_time,
440                crate::pid1::update_pid1_stats(1, locked_machine_stats.clone()),
441            );
442        }
443
444        // Run networkd collector if enabled
445        if config.networkd.enabled {
446            let config_clone = Arc::clone(&config);
447            let dbus_cell = Arc::clone(&dbus_cell);
448            let stats_clone = locked_machine_stats.clone();
449            let no_fallback = config.varlink.no_fallback;
450            spawn_timed(&mut join_set, "networkd", collect_start_time, async move {
451                if config_clone.use_varlink(&[config_clone.networkd.varlink]) {
452                    let socket_path = crate::varlink_networkd::NETWORK_SOCKET_PATH.to_string();
453                    match crate::varlink_networkd::get_networkd_state(&socket_path).await {
454                        Ok(networkd_stats) => {
455                            let mut machine_stats = stats_clone.write().await;
456                            machine_stats.networkd = networkd_stats;
457                            machine_stats.varlink_usage.networkd =
458                                Some(CollectorTransport::Varlink);
459                            return Ok(());
460                        }
461                        Err(err) => {
462                            crate::varlink_fallback::report_varlink_failure(
463                                no_fallback,
464                                "networkd",
465                                "file-based",
466                                err,
467                            )?;
468                        }
469                    }
470                }
471                stats_clone.write().await.varlink_usage.networkd = Some(CollectorTransport::Dbus);
472                // No eager connect: the cell flows into the collector and
473                // is resolved only if sysfs yields no usable map — the
474                // same lazy pattern as boot_blame.
475                crate::networkd::update_networkd_stats(
476                    config_clone.networkd.link_state_dir.clone(),
477                    None,
478                    std::path::PathBuf::from("/sys"),
479                    Some((dbus_cell, dbus_timeout)),
480                    stats_clone,
481                )
482                .await
483            });
484        }
485
486        // Run system running (SystemState) state collector
487        if config.system_state.enabled {
488            let dbus_cell = Arc::clone(&dbus_cell);
489            let stats_clone = locked_machine_stats.clone();
490            let describe = manager_describe.clone();
491            let no_fallback = config.varlink.no_fallback;
492            spawn_timed(
493                &mut join_set,
494                "system_state",
495                collect_start_time,
496                async move {
497                    if let Some(describe) = describe {
498                        match crate::varlink_system::update_system_stats(
499                            describe,
500                            stats_clone.clone(),
501                        )
502                        .await
503                        {
504                            Ok(()) => {
505                                stats_clone.write().await.varlink_usage.system_state =
506                                    Some(CollectorTransport::Varlink);
507                                return Ok(());
508                            }
509                            Err(err) => {
510                                crate::varlink_fallback::report_varlink_failure(
511                                    no_fallback,
512                                    "system state",
513                                    "D-Bus",
514                                    err,
515                                )?;
516                            }
517                        }
518                    }
519                    stats_clone.write().await.varlink_usage.system_state =
520                        Some(CollectorTransport::Dbus);
521                    let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
522                    crate::system::update_system_stats(conn, stats_clone.clone()).await
523                },
524            );
525        }
526
527        // Run service collectors if there are services listed in config
528        if config.units.enabled {
529            let config_clone = Arc::clone(&config);
530            let dbus_cell = Arc::clone(&dbus_cell);
531            let stats_clone = locked_machine_stats.clone();
532            let no_fallback = config.varlink.no_fallback;
533            spawn_timed(&mut join_set, "units", collect_start_time, async move {
534                if config_clone.use_varlink(&[config_clone.units.varlink]) {
535                    let socket_path = crate::varlink_units::METRICS_SOCKET_PATH.to_string();
536                    match crate::varlink_units::update_unit_stats(
537                        Arc::clone(&config_clone),
538                        stats_clone.clone(),
539                        socket_path,
540                    )
541                    .await
542                    {
543                        Ok(timer_names) => {
544                            // Per-service stats, timer properties and service
545                            // types all come from io.systemd.Unit.List. If that
546                            // socket is unusable the whole units collection is
547                            // redone over D-Bus: restoring pieces of it would
548                            // leave [services] and timers partially filled from
549                            // the metrics, which is worse than either path alone.
550                            if let Err(err) = crate::varlink_units::apply_unit_details(
551                                crate::varlink_unit::MANAGER_SOCKET_PATH,
552                                &stats_clone,
553                                &config_clone,
554                                "",
555                                &timer_names,
556                            )
557                            .await
558                            {
559                                crate::varlink_fallback::report_varlink_failure(
560                                    no_fallback,
561                                    "unit details",
562                                    "D-Bus",
563                                    err,
564                                )?;
565                                stats_clone.write().await.varlink_usage.units =
566                                    Some(CollectorTransport::Dbus);
567                                let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
568                                return crate::units::update_unit_stats(
569                                    config_clone,
570                                    conn,
571                                    stats_clone,
572                                    String::new(),
573                                )
574                                .await;
575                            }
576                            if config_clone.units.unit_files {
577                                let unit_files = crate::units::collect_unit_files_stats("").await;
578                                let mut ms = stats_clone.write().await;
579                                ms.units.unit_files = unit_files;
580                            }
581                            stats_clone.write().await.varlink_usage.units =
582                                Some(CollectorTransport::Varlink);
583                            return Ok(());
584                        }
585                        Err(err) => {
586                            crate::varlink_fallback::report_varlink_failure(
587                                no_fallback,
588                                "units",
589                                "D-Bus",
590                                err,
591                            )?;
592                        }
593                    }
594                }
595                stats_clone.write().await.varlink_usage.units = Some(CollectorTransport::Dbus);
596                let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
597                crate::units::update_unit_stats(config_clone, conn, stats_clone, String::new())
598                    .await
599            });
600        }
601
602        if config.machines.enabled {
603            let stats_clone = locked_machine_stats.clone();
604            let config_clone = Arc::clone(&config);
605            let dbus_cell = Arc::clone(&dbus_cell);
606            let monitord_stats_clone = locked_monitord_stats.clone();
607            let connections_clone = cached_machine_connections.clone();
608            spawn_timed(&mut join_set, "machines", collect_start_time, async move {
609                // Enumeration is D-Bus-only: machined has no varlink
610                // List API yet (see #37). Per-container collection
611                // records its own transports on the machine stats.
612                stats_clone.write().await.varlink_usage.machines = Some(CollectorTransport::Dbus);
613                let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
614                crate::machines::update_machines_stats(
615                    config_clone,
616                    conn,
617                    monitord_stats_clone,
618                    connections_clone,
619                )
620                .await
621            });
622        }
623
624        if config.dbus_stats.enabled {
625            let dbus_cell = Arc::clone(&dbus_cell);
626            let config_clone = Arc::clone(&config);
627            let stats_clone = locked_machine_stats.clone();
628            spawn_timed(
629                &mut join_set,
630                "dbus_stats",
631                collect_start_time,
632                async move {
633                    let conn = dbus_connection(&dbus_cell, dbus_timeout).await?;
634                    crate::dbus_stats::update_dbus_stats(config_clone, conn, stats_clone).await
635                },
636            );
637        }
638
639        // `verify` and `boot_blame` decide their transport internally:
640        // they take the shared cell and resolve it only on a D-Bus code
641        // path, so a varlink success or cache hit never connects.
642        if config.boot_blame.enabled {
643            let dbus_cell = Arc::clone(&dbus_cell);
644            let config_clone = Arc::clone(&config);
645            let stats_clone = locked_machine_stats.clone();
646            spawn_timed(
647                &mut join_set,
648                "boot_blame",
649                collect_start_time,
650                async move {
651                    let no_fallback = config_clone.varlink.no_fallback;
652                    crate::boot::update_boot_blame_stats(
653                        config_clone,
654                        dbus_cell,
655                        dbus_timeout,
656                        stats_clone,
657                        no_fallback,
658                    )
659                    .await
660                },
661            );
662        }
663
664        if config.verify.enabled {
665            let dbus_cell = Arc::clone(&dbus_cell);
666            let config_clone = Arc::clone(&config);
667            let stats_clone = locked_machine_stats.clone();
668            spawn_timed(&mut join_set, "verify", collect_start_time, async move {
669                crate::verify::update_verify_stats(
670                    dbus_cell,
671                    dbus_timeout,
672                    stats_clone,
673                    config_clone.verify.allowlist.clone(),
674                    config_clone.verify.blocklist.clone(),
675                    config_clone.use_varlink(&[config_clone.verify.varlink]),
676                    config_clone.varlink.no_fallback,
677                )
678                .await
679            });
680        }
681
682        if join_set.len() == 1 {
683            warn!("No collectors except systemd version scheduled to run. Exiting");
684        }
685
686        // All collectors above were spawned with `run_span` captured as their
687        // parent; the guard must be dropped before the first `.await` below
688        // since span guards are not valid to hold across an await point.
689        drop(run_guard);
690
691        // Drain join_set, collect per-collector timings + log per-collector failures
692        had_error = false;
693        let mut timings: Vec<CollectorTiming> = Vec::new();
694        while let Some(res) = join_set.join_next().await {
695            match res {
696                Ok((name, collector_result, start_offset, elapsed)) => {
697                    let success = collector_result.is_ok();
698                    if let Err(e) = collector_result {
699                        had_error = true;
700                        error!("Collector '{}' failure: {:?}", name, e);
701                    }
702                    timings.push(CollectorTiming {
703                        name,
704                        start_offset_ms: start_offset.as_secs_f64() * 1000.0,
705                        elapsed_ms: elapsed.as_secs_f64() * 1000.0,
706                        success,
707                    });
708                }
709                Err(e) => {
710                    had_error = true;
711                    error!("Join error: {:?}", e);
712                }
713            }
714        }
715
716        let elapsed_runtime = collect_start_time.elapsed();
717        let elapsed_runtime_ms = elapsed_runtime.as_millis();
718
719        // Sort timings by elapsed desc so the slowest collector is first in the JSON output
720        timings.sort_by(|a, b| {
721            b.elapsed_ms
722                .partial_cmp(&a.elapsed_ms)
723                .unwrap_or(std::cmp::Ordering::Equal)
724        });
725
726        // Per-collector lines log at debug! to keep daemon-mode noise low.
727        // The same data is on MonitordStats::collector_timings for callers that need it.
728        for t in &timings {
729            debug!(
730                "collector '{}' start_offset={:.1}ms elapsed={:.1}ms{}",
731                t.name,
732                t.start_offset_ms,
733                t.elapsed_ms,
734                if t.success { "" } else { " (FAILED)" },
735            );
736        }
737
738        {
739            // Update monitord stats with machine stats
740            let mut monitord_stats = locked_monitord_stats.write().await;
741            let machine_stats = locked_machine_stats.read().await;
742            monitord_stats.pid1 = machine_stats.pid1.clone();
743            monitord_stats.networkd = machine_stats.networkd.clone();
744            monitord_stats.system_state = machine_stats.system_state;
745            monitord_stats.version = machine_stats.version.clone();
746            monitord_stats.units = machine_stats.units.clone();
747            monitord_stats.dbus_stats = machine_stats.dbus_stats.clone();
748            monitord_stats.boot_blame = machine_stats.boot_blame.clone();
749            monitord_stats.verify_stats = machine_stats.verify_stats.clone();
750            monitord_stats.varlink_usage = machine_stats.varlink_usage.clone();
751            set_stat_collection_run_time(&mut monitord_stats, elapsed_runtime);
752            monitord_stats.collector_timings = timings;
753        }
754
755        info!("stat collection run took {}ms", elapsed_runtime_ms);
756        if output_stats {
757            let monitord_stats = locked_monitord_stats.read().await;
758            print_stats(
759                &config.monitord.key_prefix,
760                &config.monitord.output_format,
761                &monitord_stats,
762            );
763        }
764        if !config.monitord.daemon {
765            break;
766        }
767        if had_error {
768            // Drop the shared connection so the next cycle reconnects
769            // rather than reusing a bus that may have gone away mid-run.
770            // Each iteration clones the cell into its tasks at spawn time,
771            // so reassigning the local is enough for the next cycle's
772            // spawns to pick up the fresh cell.
773            dbus_cell = Arc::new(tokio::sync::OnceCell::new());
774        }
775        let sleep_time_ms = collect_interval_ms - elapsed_runtime_ms;
776        info!("stat collection sleeping for {}s 😴", sleep_time_ms / 1000);
777        tokio::time::sleep(Duration::from_millis(
778            sleep_time_ms
779                .try_into()
780                .expect("Sleep time does not fit into a u64 :O"),
781        ))
782        .await;
783    }
784    // A failed cycle returns no connection so a repeated one-shot caller
785    // reconnects instead of reusing a broken bus; a clean cycle hands the
786    // shared connection back for reuse. (In daemon mode the reset above
787    // already swapped in a fresh cell, so `get()` here is empty by design
788    // and `None` is the only honest answer.)
789    let conn = if had_error {
790        None
791    } else {
792        dbus_cell.get().cloned()
793    };
794    Ok(conn)
795}

801    /// The lazy-connect contract finding 6 on #224 pins down: a failed
802    /// connect is not cached, so consecutive failures each attempt (and
803    /// report) a fresh error rather than inheriting a poisoned cell.
804    #[tokio::test]
805    async fn test_dbus_connection_failure_is_not_cached() {
806        std::env::set_var(
807            "DBUS_SYSTEM_BUS_ADDRESS",
808            "unix:path=/nonexistent/monitord-test-bus-socket",
809        );
810        let cell: DbusCell = Arc::new(tokio::sync::OnceCell::new());
811        assert!(dbus_connection(&cell, 1).await.is_err());
812        assert!(
813            cell.get().is_none(),
814            "failed connect must not populate the cell"
815        );
816        assert!(dbus_connection(&cell, 1).await.is_err());
817        assert!(cell.get().is_none());
818    }
