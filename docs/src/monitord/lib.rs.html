49/// Per-collector timing for a single stat collection run.
50///
51/// `start_offset_ms` is the wall time between the top of the collection cycle and
52/// the moment this collector's future was first polled. A non-trivial offset
53/// indicates the spawn/scheduling loop or the runtime is delaying first poll,
54/// which means collectors are not starting in parallel as intended.
55///
56/// `elapsed_ms` is the wall time between first poll and completion.
57#[derive(serde::Serialize, serde::Deserialize, Clone, Debug, Default, PartialEq)]
58pub struct CollectorTiming {
59    /// Name of the collector (e.g. "units", "pid1", "dbus_stats")
60    pub name: String,
61    /// Milliseconds from top of the run until the spawned future's first poll.
62    /// Should be small (< a few ms) when collectors are truly running in parallel.
63    pub start_offset_ms: f64,
64    /// Milliseconds from first poll to future completion.
65    pub elapsed_ms: f64,
66    /// Whether the collector returned Ok.
67    pub success: bool,
68}

70/// Which API transport served a collector on the last run.
71///
72/// Recorded per enabled collector so operators can watch varlink adoption
73/// climb as the fleet's systemd upgrades past each endpoint's minimum
74/// version: a collector reports `Varlink` when its varlink attempt succeeded
75/// and `Dbus` when it collected the legacy way — either because varlink is
76/// disabled in config or because the varlink attempt failed and fell back.
77/// `Dbus` also covers the file-based networkd fallback: the value answers
78/// "did this collector use varlink", not "which fallback served it".
79/// Collectors with no varlink path (`pid1`, `dbus_stats`) and disabled
80/// collectors stay `None` and emit no gauge, so the present gauges are
81/// exactly the enabled set — no separate enabled-collectors counter needed.
82#[derive(
83    serde_repr::Serialize_repr,
84    serde_repr::Deserialize_repr,
85    Clone,
86    Copy,
87    Debug,
88    Default,
89    PartialEq,
90    Eq,
91)]
92#[repr(u8)]
93pub enum CollectorTransport {
94    #[default]
95    Dbus = 0,
96    Varlink = 1,
97}
98
99impl CollectorTransport {
100    /// Gauge value for flat JSON and metric exporters: 1 when varlink served
101    /// the collector, 0 for the D-Bus/file fallback.
102    ///
103    /// The `repr(u8)` discriminants are the gauge values, so every output
104    /// format (json, json-pretty, json-flat) reports the same integer —
105    /// matching the other stat enums (`SystemdSystemState`, the unit state
106    /// enums) rather than serializing as a string in some formats.
107    pub fn as_u64(self) -> u64 {
108        self as u64
109    }
110}

112/// Per-collector transport record for the last collection run.
113///
114/// Every field is `Some` when its collector ran and `None` when it did not
115/// (disabled in config). Absent-when-disabled is what makes the adoption
116/// ratio work: `count()` over the gauges is the enabled set.
117///
118/// NOTE for future collectors: the host `MachineStats` is constructed once
119/// outside the daemon loop, not fresh each iteration — so a collector with
120/// an early-return path that skips its gauge assignment would silently
121/// inherit the previous run's value instead of dropping the gauge. Always
122/// assign on every path, including cache hits and config-disabled fallbacks.
123#[derive(serde::Serialize, serde::Deserialize, Clone, Debug, Default, PartialEq)]
124pub struct VarlinkUsage {
125    /// Always collected; follows `[system-state] varlink` via the shared
126    /// `Manager.Describe` call. Normally agrees with `system_state` (one
127    /// call serves both), but persists when `[system-state]` is disabled —
128    /// making it the cheapest varlink canary on such hosts.
129    #[serde(skip_serializing_if = "Option::is_none")]
130    pub version: Option<CollectorTransport>,
131    #[serde(skip_serializing_if = "Option::is_none")]
132    pub system_state: Option<CollectorTransport>,
133    /// `Varlink` means the bulk enumeration came from the metrics stream.
134    /// Inside containers this still includes D-Bus calls underneath: the
135    /// timer backfill (`collect_all_timers_dbus`) and the oneshot type
136    /// override, which have no varlink equivalent there (see #211). The host
137    /// varlink path needs neither, so a container `1` involves strictly more
138    /// D-Bus traffic than a host `1` — compare host and container gauges
139    /// separately rather than aggregating them into one adoption ratio.
140    #[serde(skip_serializing_if = "Option::is_none")]
141    pub units: Option<CollectorTransport>,
142    #[serde(skip_serializing_if = "Option::is_none")]
143    pub networkd: Option<CollectorTransport>,
144    /// Host-side machine enumeration, still D-Bus-only until machined grows
145    /// a varlink List API (see #37). Per-container transports land on each
146    /// machine's own `MachineStats` instead.
147    #[serde(skip_serializing_if = "Option::is_none")]
148    pub machines: Option<CollectorTransport>,
149    #[serde(skip_serializing_if = "Option::is_none")]
150    pub boot_blame: Option<CollectorTransport>,
151    #[serde(skip_serializing_if = "Option::is_none")]
152    pub verify: Option<CollectorTransport>,
153}

195    pub machines: HashMap<String, MachineStats>,
196    /// Boot blame statistics: slowest units at boot with activation times in seconds
197    #[serde(skip_serializing_if = "Option::is_none")]
198    pub boot_blame: Option<boot::BootBlameStats>,
199    /// Unit verification error statistics
200    pub verify_stats: Option<verify::VerifyStats>,
201    /// End-to-end duration of the last stat collection run in milliseconds.
202    pub stat_collection_run_time_ms: f64,
203    /// Per-collector timings from the last run, sorted slowest first. Empty
204    /// before the first run completes. Callers compute parallelism ratio

208    /// Which transport served each collector on the last run.
209    pub varlink_usage: VarlinkUsage,

212/// Print statistics in the format set in configuration
213pub fn print_stats(
214    key_prefix: &str,
215    output_format: &config::MonitordOutputFormat,
216    stats: &MonitordStats,
217) {
218    match output_format {
219        config::MonitordOutputFormat::Json => println!(
220            "{}",
221            serde_json::to_string(&stats).expect("Invalid JSON serialization")
222        ),
223        config::MonitordOutputFormat::JsonFlat => println!(
224            "{}",
225            json::flatten(stats, key_prefix).expect("Invalid JSON serialization")
226        ),
227        config::MonitordOutputFormat::JsonPretty => println!(
228            "{}",
229            serde_json::to_string_pretty(&stats).expect("Invalid JSON serialization")
230        ),
231    }
232}

238/// Output produced by every spawned collector future after wrapping with timing.
239type TimedCollectorOutput = (String, anyhow::Result<()>, Duration, Duration);
240
241/// Spawn a collector future onto the join set with timing instrumentation.
242///
243/// The wrapping closure records the moment the future is first polled (relative
244/// to `collect_start`) and the elapsed wall time until it completes. Both
245/// durations and the collector name are returned alongside the original result.
246///
247/// `tokio::task::JoinSet::spawn` runs the future on a new task, and tracing
248/// spans do not cross task boundaries automatically — without explicitly
249/// capturing and re-attaching the caller's span here, every collector (and
250/// anything it spawns in turn, e.g. units.rs's per-unit tasks) would show up
251/// as an unrelated root trace instead of a child of the current collection run.
252fn spawn_timed<F>(
253    join_set: &mut tokio::task::JoinSet<TimedCollectorOutput>,
254    name: &'static str,
255    collect_start: Instant,
256    fut: F,
257) where
258    F: std::future::Future<Output = anyhow::Result<()>> + Send + 'static,
259{
260    let parent_span = tracing::Span::current();
261    let span = tracing::debug_span!(
262        parent: &parent_span,
263        "collector",
264        name,
265        elapsed_ms = tracing::field::Empty,
266        success = tracing::field::Empty,
267    );
268    let recording_span = span.clone();
269    join_set.spawn(
270        async move {
271            let task_first_poll = Instant::now();
272            let start_offset = task_first_poll.duration_since(collect_start);
273            let result = fut.await;
274            let elapsed = task_first_poll.elapsed();
275            recording_span.record("elapsed_ms", elapsed.as_secs_f64() * 1000.0);
276            recording_span.record("success", result.is_ok());
277            (name.to_string(), result, start_offset, elapsed)
278        }
279        .instrument(span),
280    );
281}

283/// Reuse an existing D-Bus connection or create a new system bus connection.
284async fn get_or_create_dbus_connection(
285    config: &config::Config,
286    maybe_connection: Option<zbus::Connection>,
287) -> Result<zbus::Connection, MonitordError> {
288    match maybe_connection {
289        Some(conn) => Ok(conn),
290        None => Ok(zbus::connection::Builder::system()?
291            .method_timeout(std::time::Duration::from_secs(config.monitord.dbus_timeout))
292            .build()
293            .await?),
294    }
295}

297/// Main statistic collection function running what's required by configuration in parallel
298/// Takes an optional locked stats struct to update and to output stats to STDOUT or not.
299/// Takes an optional D-Bus connection. Returns `Some(connection)` if the
300/// collection cycle completed without errors (meaning the connection is reusable),
301/// `None` if errors occurred.
302pub async fn stat_collector(
303    config: config::Config,
304    maybe_locked_stats: Option<Arc<RwLock<MonitordStats>>>,
305    output_stats: bool,
306    maybe_connection: Option<zbus::Connection>,
307) -> Result<Option<zbus::Connection>, MonitordError> {
308    let mut collect_interval_ms: u128 = 0;
309    if config.monitord.daemon {
310        collect_interval_ms = (config.monitord.daemon_stats_refresh_secs * 1000).into();
311    }
312
313    let config = Arc::new(config);
314    let locked_monitord_stats: Arc<RwLock<MonitordStats>> =
315        maybe_locked_stats.unwrap_or(Arc::new(RwLock::new(MonitordStats::default())));
316    let locked_machine_stats: Arc<RwLock<MachineStats>> =
317        Arc::new(RwLock::new(MachineStats::default()));
318    let cached_machine_connections: Arc<tokio::sync::Mutex<machines::MachineConnections>> =
319        Arc::new(tokio::sync::Mutex::new(HashMap::new()));
320    std::env::set_var("DBUS_SYSTEM_BUS_ADDRESS", &config.monitord.dbus_address);
321    let sdc = get_or_create_dbus_connection(&config, maybe_connection).await?;
322    let mut join_set: tokio::task::JoinSet<TimedCollectorOutput> = tokio::task::JoinSet::new();
323    let mut had_error;
324
325    loop {
326        let collect_start_time = Instant::now();
327        // Kept alive for the whole iteration so its reported duration covers
328        // the full run (spawn + drain), not just the synchronous spawn phase
329        // below where `run_guard` is held.
330        let run_span = tracing::info_span!("stat_collector_run");
331        let run_guard = run_span.enter();
332        info!("Starting stat collection run");
333
334        // One Manager.Describe serves both the version and system state
335        // collectors: PID 1 handles varlink requests one at a time, so a second
336        // call would sit behind the units collector's whole metrics stream.
337        //
338        // Which request reaches PID 1 first is still up to the scheduler, so
339        // under CPU pressure this call can land behind that stream anyway. That
340        // is the old cost for one call rather than two, not a new failure mode.
341        // Making it deterministic would mean holding the units collector until
342        // this resolves, which would put the collector that gates the cycle
343        // behind an unrelated socket.
344        let manager_describe = config.use_varlink(&[config.system_state.varlink]).then(|| {
345            crate::varlink_system::shared_describe(
346                crate::varlink_system::MANAGER_SOCKET_PATH.to_string(),
347            )
348        });
349
350        // Always collect systemd version
351        {
352            let sdc_clone = sdc.clone();
353            let stats_clone = locked_machine_stats.clone();
354            let describe = manager_describe.clone();
355            spawn_timed(&mut join_set, "version", collect_start_time, async move {
356                if let Some(describe) = describe {
357                    match crate::varlink_system::update_version(describe, stats_clone.clone()).await
358                    {
359                        Ok(()) => {
360                            stats_clone.write().await.varlink_usage.version =
361                                Some(CollectorTransport::Varlink);
362                            return Ok(());
363                        }
364                        Err(err) => {
365                            warn!("Varlink version failed, falling back to D-Bus: {:?}", err);
366                        }
367                    }
368                }
369                stats_clone.write().await.varlink_usage.version = Some(CollectorTransport::Dbus);
370                crate::system::update_version(sdc_clone, stats_clone.clone()).await
371            });
372        }
373
374        // Collect pid1 procfs stats
375        if config.pid1.enabled {
376            spawn_timed(
377                &mut join_set,
378                "pid1",
379                collect_start_time,
380                crate::pid1::update_pid1_stats(1, locked_machine_stats.clone()),
381            );
382        }
383
384        // Run networkd collector if enabled
385        if config.networkd.enabled {
386            let config_clone = Arc::clone(&config);
387            let sdc_clone = sdc.clone();
388            let stats_clone = locked_machine_stats.clone();
389            spawn_timed(&mut join_set, "networkd", collect_start_time, async move {
390                if config_clone.use_varlink(&[config_clone.networkd.varlink]) {
391                    let socket_path = crate::varlink_networkd::NETWORK_SOCKET_PATH.to_string();
392                    match crate::varlink_networkd::get_networkd_state(&socket_path).await {
393                        Ok(networkd_stats) => {
394                            let mut machine_stats = stats_clone.write().await;
395                            machine_stats.networkd = networkd_stats;
396                            machine_stats.varlink_usage.networkd =
397                                Some(CollectorTransport::Varlink);
398                            return Ok(());
399                        }
400                        Err(err) => {
401                            warn!(
402                                "Varlink networkd stats failed, falling back to file-based: {:?}",
403                                err
404                            );
405                        }
406                    }
407                }
408                stats_clone.write().await.varlink_usage.networkd = Some(CollectorTransport::Dbus);
409                crate::networkd::update_networkd_stats(
410                    config_clone.networkd.link_state_dir.clone(),
411                    None,
412                    sdc_clone,
413                    stats_clone,
414                )
415                .await
416            });
417        }
418
419        // Run system running (SystemState) state collector
420        if config.system_state.enabled {
421            let sdc_clone = sdc.clone();
422            let stats_clone = locked_machine_stats.clone();
423            let describe = manager_describe.clone();
424            spawn_timed(
425                &mut join_set,
426                "system_state",
427                collect_start_time,
428                async move {
429                    if let Some(describe) = describe {
430                        match crate::varlink_system::update_system_stats(
431                            describe,
432                            stats_clone.clone(),
433                        )
434                        .await
435                        {
436                            Ok(()) => {
437                                stats_clone.write().await.varlink_usage.system_state =
438                                    Some(CollectorTransport::Varlink);
439                                return Ok(());
440                            }
441                            Err(err) => {
442                                warn!(
443                                    "Varlink system state failed, falling back to D-Bus: {:?}",
444                                    err
445                                );
446                            }
447                        }
448                    }
449                    stats_clone.write().await.varlink_usage.system_state =
450                        Some(CollectorTransport::Dbus);
451                    crate::system::update_system_stats(sdc_clone, stats_clone.clone()).await
452                },
453            );
454        }
455
456        // Run service collectors if there are services listed in config
457        if config.units.enabled {
458            let config_clone = Arc::clone(&config);
459            let sdc_clone = sdc.clone();
460            let stats_clone = locked_machine_stats.clone();
461            spawn_timed(&mut join_set, "units", collect_start_time, async move {
462                if config_clone.use_varlink(&[config_clone.units.varlink]) {
463                    let socket_path = crate::varlink_units::METRICS_SOCKET_PATH.to_string();
464                    match crate::varlink_units::update_unit_stats(
465                        Arc::clone(&config_clone),
466                        stats_clone.clone(),
467                        socket_path,
468                    )
469                    .await
470                    {
471                        Ok(timer_names) => {
472                            // Per-service stats, timer properties and service
473                            // types all come from io.systemd.Unit.List. If that
474                            // socket is unusable the whole units collection is
475                            // redone over D-Bus: restoring pieces of it would
476                            // leave [services] and timers partially filled from
477                            // the metrics, which is worse than either path alone.
478                            if let Err(err) = crate::varlink_units::apply_unit_details(
479                                crate::varlink_unit::MANAGER_SOCKET_PATH,
480                                &stats_clone,
481                                &config_clone,
482                                "",
483                                &timer_names,
484                            )
485                            .await
486                            {
487                                warn!(
488                                    "Varlink unit details failed, falling back to D-Bus: {:?}",
489                                    err
490                                );
491                                stats_clone.write().await.varlink_usage.units =
492                                    Some(CollectorTransport::Dbus);
493                                return crate::units::update_unit_stats(
494                                    config_clone,
495                                    sdc_clone,
496                                    stats_clone,
497                                    String::new(),
498                                )
499                                .await;
500                            }
501                            if config_clone.units.unit_files {
502                                let unit_files = crate::units::collect_unit_files_stats("").await;
503                                let mut ms = stats_clone.write().await;
504                                ms.units.unit_files = unit_files;
505                            }
506                            stats_clone.write().await.varlink_usage.units =
507                                Some(CollectorTransport::Varlink);
508                            return Ok(());
509                        }
510                        Err(err) => {
511                            warn!(
512                                "Varlink units stats failed, falling back to D-Bus: {:?}",
513                                err
514                            );
515                        }
516                    }
517                }
518                stats_clone.write().await.varlink_usage.units = Some(CollectorTransport::Dbus);
519                crate::units::update_unit_stats(config_clone, sdc_clone, stats_clone, String::new())
520                    .await
521            });
522        }
523
524        if config.machines.enabled {
525            let stats_clone = locked_machine_stats.clone();
526            let config_clone = Arc::clone(&config);
527            let sdc_clone = sdc.clone();
528            let monitord_stats_clone = locked_monitord_stats.clone();
529            let connections_clone = cached_machine_connections.clone();
530            spawn_timed(&mut join_set, "machines", collect_start_time, async move {
531                // Enumeration is D-Bus-only: machined has no varlink
532                // List API yet (see #37). Per-container collection
533                // records its own transports on the machine stats.
534                stats_clone.write().await.varlink_usage.machines = Some(CollectorTransport::Dbus);
535                crate::machines::update_machines_stats(
536                    config_clone,
537                    sdc_clone,
538                    monitord_stats_clone,
539                    connections_clone,
540                )
541                .await
542            });
543        }
544
545        if config.dbus_stats.enabled {
546            spawn_timed(
547                &mut join_set,
548                "dbus_stats",
549                collect_start_time,
550                crate::dbus_stats::update_dbus_stats(
551                    Arc::clone(&config),
552                    sdc.clone(),
553                    locked_machine_stats.clone(),
554                ),
555            );
556        }
557
558        if config.boot_blame.enabled {
559            spawn_timed(
560                &mut join_set,
561                "boot_blame",
562                collect_start_time,
563                crate::boot::update_boot_blame_stats(
564                    Arc::clone(&config),
565                    sdc.clone(),
566                    locked_machine_stats.clone(),
567                ),
568            );
569        }
570
571        if config.verify.enabled {
572            spawn_timed(
573                &mut join_set,
574                "verify",
575                collect_start_time,
576                crate::verify::update_verify_stats(
577                    sdc.clone(),
578                    locked_machine_stats.clone(),
579                    config.verify.allowlist.clone(),
580                    config.verify.blocklist.clone(),
581                    config.use_varlink(&[config.verify.varlink]),
582                ),
583            );
584        }
585
586        if join_set.len() == 1 {
587            warn!("No collectors except systemd version scheduled to run. Exiting");
588        }
589
590        // All collectors above were spawned with `run_span` captured as their
591        // parent; the guard must be dropped before the first `.await` below
592        // since span guards are not valid to hold across an await point.
593        drop(run_guard);
594
595        // Drain join_set, collect per-collector timings + log per-collector failures
596        had_error = false;
597        let mut timings: Vec<CollectorTiming> = Vec::new();
598        while let Some(res) = join_set.join_next().await {
599            match res {
600                Ok((name, collector_result, start_offset, elapsed)) => {
601                    let success = collector_result.is_ok();
602                    if let Err(e) = collector_result {
603                        had_error = true;
604                        error!("Collector '{}' failure: {:?}", name, e);
605                    }
606                    timings.push(CollectorTiming {
607                        name,
608                        start_offset_ms: start_offset.as_secs_f64() * 1000.0,
609                        elapsed_ms: elapsed.as_secs_f64() * 1000.0,
610                        success,
611                    });
612                }
613                Err(e) => {
614                    had_error = true;
615                    error!("Join error: {:?}", e);
616                }
617            }
618        }
619
620        let elapsed_runtime = collect_start_time.elapsed();
621        let elapsed_runtime_ms = elapsed_runtime.as_millis();
622
623        // Sort timings by elapsed desc so the slowest collector is first in the JSON output
624        timings.sort_by(|a, b| {
625            b.elapsed_ms
626                .partial_cmp(&a.elapsed_ms)
627                .unwrap_or(std::cmp::Ordering::Equal)
628        });
629
630        // Per-collector lines log at debug! to keep daemon-mode noise low.
631        // The same data is on MonitordStats::collector_timings for callers that need it.
632        for t in &timings {
633            debug!(
634                "collector '{}' start_offset={:.1}ms elapsed={:.1}ms{}",
635                t.name,
636                t.start_offset_ms,
637                t.elapsed_ms,
638                if t.success { "" } else { " (FAILED)" },
639            );
640        }
641
642        {
643            // Update monitord stats with machine stats
644            let mut monitord_stats = locked_monitord_stats.write().await;
645            let machine_stats = locked_machine_stats.read().await;
646            monitord_stats.pid1 = machine_stats.pid1.clone();
647            monitord_stats.networkd = machine_stats.networkd.clone();
648            monitord_stats.system_state = machine_stats.system_state;
649            monitord_stats.version = machine_stats.version.clone();
650            monitord_stats.units = machine_stats.units.clone();
651            monitord_stats.dbus_stats = machine_stats.dbus_stats.clone();
652            monitord_stats.boot_blame = machine_stats.boot_blame.clone();
653            monitord_stats.verify_stats = machine_stats.verify_stats.clone();
654            monitord_stats.varlink_usage = machine_stats.varlink_usage.clone();
655            set_stat_collection_run_time(&mut monitord_stats, elapsed_runtime);
656            monitord_stats.collector_timings = timings;
657        }
658
659        info!("stat collection run took {}ms", elapsed_runtime_ms);
660        if output_stats {
661            let monitord_stats = locked_monitord_stats.read().await;
662            print_stats(
663                &config.monitord.key_prefix,
664                &config.monitord.output_format,
665                &monitord_stats,
666            );
667        }
668        if !config.monitord.daemon {
669            break;
670        }
671        let sleep_time_ms = collect_interval_ms - elapsed_runtime_ms;
672        info!("stat collection sleeping for {}s 😴", sleep_time_ms / 1000);
673        tokio::time::sleep(Duration::from_millis(
674            sleep_time_ms
675                .try_into()
676                .expect("Sleep time does not fit into a u64 :O"),
677        ))
678        .await;
679    }
680    Ok(if had_error { None } else { Some(sdc) })
681}
